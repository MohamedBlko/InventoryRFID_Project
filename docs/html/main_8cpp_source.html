147String extractTagSku(const String& json) {
148 int idIndex = json.indexOf("\"id\":\"");
149 if (idIndex == -1) return "";
150 int start = idIndex + 6; // length of '"id":"'
151 int end = json.indexOf("\"", start);
152 if (end == -1) return "";
153 String sku = json.substring(start, end);
154 // Remove any trailing null bytes or whitespace
155 sku.trim();
156 return sku;

159String readStringFromUltralight(byte startPage, byte length) {

170
171 for (int j = 0; j < 4 && result.length() < length; j++) {
172 byte b = rawBuf[j];
173 if (b == 0x00) break;
174 result += (char)b;
175 }
176 }

178 return extractTagSku(result);
